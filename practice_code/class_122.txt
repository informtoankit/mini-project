name: Complete GitHub Actions Learning Workflow

on:
  push:
    branches:
      - main
  pull_request:
  workflow_dispatch:

# Workflow Level Environment Variables
env:
  APP_NAME: sample
  STAGE: dev

jobs:
  tooling:
    runs-on: ubuntu-latest

    steps:
      # Checkout Repository
      - name: Checkout Repository
        uses: actions/checkout@v4

      # Print GitHub Context
      - name: Print GitHub Context
        run: echo '${{ toJson(github) }}'

      # Print Job Context
      - name: Print Job Context
        run: echo '${{ toJson(job) }}'

      # Workflow Variables
      - name: Print Workflow Variables
        run: |
          echo "APP_NAME=${{ env.APP_NAME }}"
          echo "STAGE=${{ env.STAGE }}"
          echo "Triggered By: $GITHUB_ACTOR"

      # Default GitHub Environment Variables
      - name: Default GitHub Environment Variables
        run: |
          echo "GITHUB_ACTION: ${GITHUB_ACTION}"
          echo "HOME: ${HOME}"
          echo "GITHUB_WORKSPACE: ${GITHUB_WORKSPACE}"
          echo "GITHUB_ACTIONS: ${GITHUB_ACTIONS}"
          echo "GITHUB_API_URL: ${GITHUB_API_URL}"
          echo "GITHUB_EVENT_PATH: ${GITHUB_EVENT_PATH}"
          echo "GITHUB_JOB: ${GITHUB_JOB}"
          echo "GITHUB_REPOSITORY: ${GITHUB_REPOSITORY}"
          echo "GITHUB_SHA: ${GITHUB_SHA}"
          echo "GITHUB_REF: ${GITHUB_REF}"
          echo "GITHUB_ACTOR: ${GITHUB_ACTOR}"
          echo "RUNNER_OS: ${RUNNER_OS}"

      # Show Workspace Information
      - name: Show Workspace Information
        run: |
          echo "===== CURRENT DIRECTORY ====="
          pwd

          echo "===== FILES ====="
          ls -la

          echo "===== GITHUB VARIABLES ====="
          echo "GITHUB_SHA=$GITHUB_SHA"
          echo "GITHUB_REPOSITORY=$GITHUB_REPOSITORY"
          echo "GITHUB_WORKSPACE=$GITHUB_WORKSPACE"
          echo "GITHUB_REF=$GITHUB_REF"
          echo "GITHUB_ACTOR=$GITHUB_ACTOR"

      # Create Custom Variable
      - name: Create SHORT_SHA Variable
        run: |
          echo "SHORT_SHA=${GITHUB_SHA:0:7}" >> $GITHUB_ENV

      # Use Custom Variable
      - name: Print SHORT_SHA
        run: |
          echo "Commit Short SHA = $SHORT_SHA"

      # Show Operating System
      - name: Show OS
        run: uname -a

      # Node.js and npm
      - name: Node & npm
        run: |
          node -v
          npm -v

      # Python and pip
      - name: Python & pip
        run: |
          python3 --version
          pip3 --version

      # Go
      - name: Go Version
        run: go version

      # Docker and kubectl
      - name: Docker & kubectl
        run: |
          docker --version
          kubectl version --client=true

  hello:
    runs-on: ubuntu-latest

    # Job Level Environment Variables
    env:
      MEMORY: TOKYO

    steps:
      - name: Custom Environment Variables
        env:
          SUBSCRIBE: SUPERMAN
          NAME: ANKIT
        run: |
          echo "MEMORY=$MEMORY"
          echo "SUBSCRIBE=$SUBSCRIBE"
          echo "NAME=$NAME"

      - name: Environment Variable Priority Demo
        env:
          MEMORY: DELHI
        run: |
          echo "Step Level MEMORY=$MEMORY"
